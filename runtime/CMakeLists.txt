cmake_minimum_required(VERSION 3.20)
project(PanelGlassRuntime LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick QuickWidgets Qml)
find_package(KF6ColorScheme REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6KirigamiPlatform REQUIRED)
add_library(panelglass MODULE plugin.cpp)
target_link_libraries(panelglass PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::QuickWidgets Qt6::Qml KF6::ColorScheme KF6::WindowSystem KF6::KirigamiPlatform)
set_target_properties(panelglass PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/generic")
option(PANELGLASS_TESTS "Build functional runtime checks" OFF)
if(PANELGLASS_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(panelglass-test runtime-test.cpp)
    target_link_libraries(panelglass-test PRIVATE Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::Test)
endif()
